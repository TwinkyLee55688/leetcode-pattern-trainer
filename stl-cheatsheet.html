2. Visual Memory — Don't Memorize Blindly

Stack — LIFO

        push(x)
           ↓
        ┌─────┐
 top() →│  C  │← pop()
        ├─────┤
        │  B  │
        ├─────┤
        │  A  │
        └─────┘

Last In → First Out

Think: a stack of plates.

Queue — FIFO

pop() ← [ A ][ B ][ C ][ D ] ← push()
          ↑               ↑
       front()           back()

First In → First Out

push() → BACK
pop()  ← FRONT

back() means “look at the back”. It does NOT mean queue exposes push_back().

Deque — Both Ends

push_front()
      ↓
      [ A ][ B ][ C ]
        ↑         ↑
     front()    back()
        ↓         ↓
 pop_front()   pop_back()
                  ↑
             push_back()

Queue restricts you. Deque gives both ends.

Priority Queue — Heap

          top()
            ↓
          [ 9 ]
         /     \
       [ 7 ]   [ 5 ]
       /   \
     [ 2 ] [ 3 ]

priority_queue<int>
        ↓
     MAX HEAP

top() = highest priority, not oldest element.

Vector — Dynamic Array

index    0     1     2     3
         ↓     ↓     ↓     ↓
       [ A ][ B ][ C ][ D ]
         ↑               ↑
      front()           back()
                          ↑
                     push_back()
                     pop_back()

Random access:
v[i]

Pair — Travel Together

pair<string,int>

┌──────────────────┐
│  "word"  │   3   │
└──────────────────┘
      ↑         ↑
    first     second

auto [word, steps] = p;

Useful when one object needs metadata.

4. Interview Patterns — Structure → Code
BFS → queue 
queue<pair<int,int>> q;
unordered_set<int> visited;

q.push({start, 0});
visited.insert(start);

while (!q.empty()) {

    auto [node, dist] = q.front();
    q.pop();

    if (node == target)
        return dist;

    for (int nei : graph[node]) {

        if (!visited.count(nei)) {

            visited.insert(nei);

            q.push({
                nei,
                dist + 1
            });
        }
    }
}

Queue preserves level / distance order.

Word Mutation BFS 
unordered_set<string> dict(
    wordList.begin(),
    wordList.end()
);

queue<pair<string,int>> q;

q.push({beginWord, 1});


while (!q.empty()) {

    auto [word, steps] = q.front();
    q.pop();

    if (word == endWord)
        return steps;

    for (int i = 0;
         i < word.size();
         i++) {

        string next = word;

        for (char c = 'a';
             c <= 'z';
             c++) {

            next[i] = c;

            if (dict.count(next)) {

                dict.erase(next);

                q.push({
                    next,
                    steps + 1
                });
            }
        }
    }
}

Mutation problem → implicit graph → BFS.

Frequency → unordered_map 
unordered_map<int,int> freq;

for (int x : nums) {
    freq[x]++;
}


for (auto& [value, count] : freq) {

    cout << value
         << " appears "
         << count
         << " times";
}

Need value → count? Think hash map.

Dijkstra → Min Heap 
using P = pair<int,int>;
// {distance, node}

priority_queue<
    P,
    vector<P>,
    greater<P>
> pq;

vector<int> dist(n, INT_MAX);

dist[start] = 0;
pq.push({0, start});


while (!pq.empty()) {

    auto [d, node] = pq.top();
    pq.pop();

    if (d > dist[node])
        continue;

    for (auto [nei, weight]
         : graph[node]) {

        int nd = d + weight;

        if (nd < dist[nei]) {

            dist[nei] = nd;

            pq.push({
                nd,
                nei
            });
        }
    }
}

Need the smallest current distance → min heap.

Next Greater → Monotonic Stack 
stack<int> st;
// store indices

for (int i = 0;
     i < nums.size();
     i++) {

    while (!st.empty() &&
           nums[st.top()] < nums[i]) {

        int j = st.top();
        st.pop();

        ans[j] = i - j;
    }

    st.push(i);
}

Stack often stores indices, not values.

Sliding Window Maximum → deque 
deque<int> dq;
// store indices

for (int i = 0;
     i < nums.size();
     i++) {

    // Remove expired index
    if (!dq.empty() &&
        dq.front() <= i - k) {

        dq.pop_front();
    }


    // Maintain decreasing values
    while (!dq.empty() &&
           nums[dq.back()] <= nums[i]) {

        dq.pop_back();
    }


    dq.push_back(i);


    if (i >= k - 1) {

        ans.push_back(
            nums[dq.front()]
        );
    }
}

Need both ends → deque, not queue.

Weighted Graph → pair 
vector<
    vector<pair<int,int>>
> graph(n);


// edge: u → v, weight w
graph[u].push_back({v, w});


for (auto [nei, weight]
     : graph[u]) {

    cout << nei
         << " "
         << weight;
}

Neighbor and weight must travel together.

Level Order → queue 
queue<TreeNode*> q;

q.push(root);


while (!q.empty()) {

    int size = q.size();

    for (int i = 0;
         i < size;
         i++) {

        TreeNode* node = q.front();
        q.pop();

        if (node->left)
            q.push(node->left);

        if (node->right)
            q.push(node->right);
    }

    // One whole level finished here
}

Snapshot q.size() before processing a level.

5. Interview Gotchas
1. pop() removes — it does NOT return. 
// Queue
int x = q.front();
q.pop();


// Stack
int x = st.top();
st.pop();


// Heap
int x = pq.top();
pq.pop();
2. mp[key] can INSERT a missing key. 
unordered_map<string,int> mp;


// This may create "apple"
int x = mp["apple"];


// Only checking existence?
if (mp.count("apple")) {
    ...
}


// Or
if (mp.find("apple") != mp.end()) {
    ...
}
3. queue is intentionally restrictive. 
queue<int> q;

q.push(x);       // add BACK

q.front();       // peek FRONT
q.back();        // peek BACK

q.pop();         // remove FRONT


// NOT AVAILABLE:
q.push_front(x); // ❌
q.pop_back();    // ❌

A queue is a container adapter. It intentionally hides operations that could violate FIFO.

4. priority_queue is MAX heap by default. 
// Largest element first
priority_queue<int> maxHeap;


// Smallest element first
priority_queue<
    int,
    vector<int>,
    greater<int>
> minHeap;
5. Insert into visited when ENQUEUING. 
if (!visited.count(nei)) {

    visited.insert(nei);

    q.push(nei);
}

Don't wait until pop(), or the same node may be enqueued many times.

6. 10-Second Recall 
// VECTOR
vector<int> v(n, 0);
v.push_back(x);
v[i];
v.pop_back();


// STACK
stack<int> st;
st.push(x);
st.top();
st.pop();


// QUEUE
queue<int> q;
q.push(x);        // BACK
q.front();
q.back();
q.pop();          // FRONT


// DEQUE
deque<int> dq;
dq.push_front(x);
dq.push_back(x);
dq.front();
dq.back();
dq.pop_front();
dq.pop_back();


// MAX HEAP
priority_queue<int> maxHeap;


// MIN HEAP
priority_queue<
    int,
    vector<int>,
    greater<int>
> minHeap;


// SET
unordered_set<int> seen;
seen.insert(x);
seen.count(x);
seen.erase(x);


// MAP
unordered_map<int,int> mp;
mp[key] = value;
mp.count(key);
mp.erase(key);


// PAIR
pair<int,int> p = {node, dist};

auto [node, dist] = p;
